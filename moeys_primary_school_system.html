const FOLDER_ID = "1GCMdTew9rgw5lwkBhmsEuy8WBGELNM1g";
const FILE_NAME = "phnom_pom_primary_school_db.json";

function doGet(e) {
  try {
    const folder = DriveApp.getFolderById(FOLDER_ID);
    const files = folder.getFilesByName(FILE_NAME);
    if (files.hasNext()) {
      const file = files.next();
      const content = file.getBlob().getDataAsString();
      return ContentService.createTextOutput(content).setMimeType(ContentService.MimeType.JSON);
    } else {
      return ContentService.createTextOutput(JSON.stringify({ status: "empty" })).setMimeType(ContentService.MimeType.JSON);
    }
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({ error: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}

function doPost(e) {
  try {
    const jsonData = e.postData.contents;
    const folder = DriveApp.getFolderById(FOLDER_ID);
    const files = folder.getFilesByName(FILE_NAME);
    let file;
    if (files.hasNext()) {
      file = files.next();
      file.setContent(jsonData);
    } else {
      file = folder.createFile(FILE_NAME, jsonData, MimeType.PLAIN_TEXT);
    }

    try {
      const data = JSON.parse(jsonData);
      updateGoogleSpreadsheet(folder, data);
    } catch(sheetErr) {}

    return ContentService.createTextOutput(JSON.stringify({ status: "success", updated: new Date().toISOString() })).setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({ error: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}

function updateGoogleSpreadsheet(folder, data) {
  const SPREADSHEET_NAME = "សាលាបឋមសិក្សាភ្នំពុំ_តារាងទិន្នន័យសរុប";
  const files = folder.getFilesByName(SPREADSHEET_NAME);
  let ss;
  if (files.hasNext()) {
    ss = SpreadsheetApp.open(files.next());
  } else {
    ss = SpreadsheetApp.create(SPREADSHEET_NAME);
    const file = DriveApp.getFileById(ss.getId());
    folder.addFile(file);
    DriveApp.getRootFolder().removeFile(file);
  }

  if (data.students && data.students.length > 0) {
    let sheet = ss.getSheetByName("បញ្ជីសិស្ស") || ss.insertSheet("បញ្ជីសិស្ស");
    sheet.clear();
    const rows = [["ល.រ", "អត្តលេខ", "គោត្តនាម-នាម", "Name Latin", "ភេទ", "ថ្ងៃខែឆ្នាំកំណើត", "ថ្នាក់", "ឆ្នាំសិក្សា", "អាណាព្យាបាល", "ទូរស័ព្ទ"]];
    data.students.forEach((s, i) => {
      rows.push([i + 1, s.code, s.nameKhmer, s.nameLatin, s.gender === 'F' ? 'ស្រី' : 'ប្រុស', s.dob, 'ថ្នាក់ទី ' + s.grade + s.section, s.academicYear, s.guardianName, s.guardianPhone]);
    });
    sheet.getRange(1, 1, rows.length, rows[0].length).setValues(rows);
    sheet.getRange(1, 1, 1, rows[0].length).setFontWeight("bold").setBackground("#d9e1f2");
  }

  if (data.scores && data.scores.length > 0) {
    let sheet = ss.getSheetByName("ពិន្ទុសិស្ស") || ss.insertSheet("ពិន្ទុសិស្ស");
    sheet.clear();
    const rows = [["ល.រ", "អត្តលេខ", "ឈ្មោះសិស្ស", "ភេទ", "អំណាន", "សំណេរ", "គណិត", "វិទ្យាសាស្ត្រ", "សង្គម", "សីលធម៌", "សិល្បៈ", "សរុប", "មធ្យមភាគ"]];
    data.scores.forEach((sc, i) => {
      const tot = (sc.reading + sc.writing + sc.math + sc.science + sc.social + sc.moral + sc.arts).toFixed(1);
      const avg = (tot / 7).toFixed(2);
      rows.push([i + 1, sc.studentCode, sc.studentName, sc.gender === 'F' ? 'ស្រី' : 'ប្រុស', sc.reading, sc.writing, sc.math, sc.science, sc.social, sc.moral, sc.arts, tot, avg]);
    });
    sheet.getRange(1, 1, rows.length, rows[0].length).setValues(rows);
    sheet.getRange(1, 1, 1, rows[0].length).setFontWeight("bold").setBackground("#d9e1f2");
  }
}
